def saveData():
    dBJson.write_text(json.dumps(data))

    return {"status": "ok"}


def isUsernameTaken(username):
    if username in data["accounts"].keys():
        return True

    return False


def verifyInput(username, password):
    if len(username) > 15 or len(username) < 3:
        return {
            "status": "declined",
            "msg": "Username too long or too short make it between 3 and 15 characters",
        }

    for i in username.lower():
        if i not in usernameChars:
            return {
                "status": "declined",
                "msg": "Invalid Characters use only letters numbers and _",
            }

    try:
        if data["accounts"][username][0] == password:
            return {"status": "ok", "msg": "Login Successful"}
    except KeyError:
        pass
    return {"status": "declined", "msg": "Login Failed, incorrect password"}


def createAccount(username, password):
    # Check if account exists (to avoid account reseting / stealing usernames)
    if username in data["accounts"].keys():
        return {"status": "declined", "msg": "Account already exists"}
    # Create account in format Username: [password, points]
    data["accounts"][username] = [password, 0]
    #
    saveData()
    #
    return {"status": "ok"}


def addPointsToAccount(username, points):
    # Check if account exists
    if username not in data["accounts"].keys():
        return {"status": "declined", "msg": f'Username "{username}" not found'}
    #
    data["accounts"][username][1] += points
    #
    saveData()
    #
    return {"status": "ok"}
